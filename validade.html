function validarQuantidade(quantidade, estoqueDisponivel = Infinity) { // Converte para número caso venha de um input de texto const qtd = Number(quantidade); // 1. Verifica se é um número válido e inteiro if (isNaN(qtd) || !Number.isInteger(qtd)) { return { valido: false, erro: "A quantidade deve ser um número inteiro." }; } // 2. Verifica se é maior que zero if (qtd <= 0) { return { valido: false, erro: "A quantidade deve ser maior que zero." }; } // 3. Verifica se há estoque suficiente (opcional) if (qtd > estoqueDisponivel) { return { valido: false, erro: `Estoque insuficiente. Disponível: ${estoqueDisponivel}` }; } // Se passar em tudo return { valido: true, quantidade: qtd }; } // --- Exemplos de uso --- console.log(validarQuantidade(3)); // { valido: true, quantidade: 3 } console.log(validarQuantidade("5")); // { valido: true, quantidade: 5 } console.log(validarQuantidade(2.5)); // { valido: false, erro: "A quantidade..." } console.log(validarQuantidade(-1)); // { valido: false, erro: "A quantidade deve ser maior que zero." } console.log(validarQuantidade(10, 5)); // { valido: false, erro: "Estoque insuficiente. Disponível: 5" } const inputQtd = document.getElementById('quantidade-produto'); const estoqueAtual = 10; // Exemplo de limite function finalizarCompra() { const validacao = validarQuantidade(inputQtd.value, estoqueAtual); if (!validacao.valido) { alert(validacao.erro); inputQtd.focus(); return; } console.log(`Sucesso! Adicionando ${validacao.quantidade} itens ao carrinho.`); }
