Validação de Quantidade if (item.quantidade <= 0) { return { valido: false, motivo: "Produto fora de estoque." }; } // 2. Validação de Validade if (dataValidade < dataAtual) { return { valido: false, motivo: "Produto vencido." }; } return { valido: true, motivo: "Produto pronto para venda!" }; } // Executando o teste const resultado = validarProduto(produto); console.log(resultado.motivo);
